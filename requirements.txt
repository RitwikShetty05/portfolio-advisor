# AI Portfolio Advisory System — Python dependencies
# Tested on Python 3.10+ and macOS arm64 (Apple M1/M2).
# Install with:  pip install -r requirements.txt

# Core data & numerics
# numpy capped <2.3: hmmlearn 0.3.3 ships a compiled wheel built against the
# numpy ≤2.2 ABI, and importing it under numpy 2.3 throws an ImportError on
# hosted boxes (this took the live app down once). Cap keeps the resolver on
# the proven-compatible pair (numpy 2.1/2.2 + hmmlearn 0.3.3).
pandas>=2.0.0
numpy>=1.24.0,<2.3
scipy>=1.10.0

# Market data
yfinance>=0.2.40

# Technical indicators
ta>=0.11.0

# Machine learning
# hmmlearn pinned exactly: it's the one compiled dependency, and an
# unpinned >= let the resolver drift into a numpy-ABI-incompatible state on
# a Streamlit Cloud redeploy. 0.3.3 is the proven-good version (paired with
# the numpy<2.3 cap above). If hmmlearn ever fails to import anyway, the app
# degrades to the KMeans regime method instead of crashing (see regime.py).
scikit-learn>=1.3.0
hmmlearn==0.3.3

# Visualisation
plotly>=5.18.0
matplotlib>=3.7.0
seaborn>=0.12.0

# Frontend — ≥1.37 for st.fragment(run_every=…), which powers the
# auto-refreshing status bar + ticker tape (feature-detected, so the app
# still runs — statically — on older versions).
streamlit>=1.37.0

# Portfolio uploads — Excel + PDF parsing
openpyxl>=3.1.0
pdfplumber>=0.10.0
